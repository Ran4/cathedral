```rust
/// The two frozen salience inputs, computed once wherever a `Fact` is born.
///
/// Called by `FactCatalog::seed` (the authored path) and by `mint::install_fact`
/// (both coded mints), and by nothing else — a second caller is a second answer.
pub(crate) fn frozen_ears(world: &World, subject: &[ActorId]) -> (BTreeSet<ActorId>, Option<String>) {
    let mut quiet: BTreeSet<ActorId> = subject.iter().cloned().collect();
    for who in subject {
        if let Some(lore) = world.characters.get(who).and_then(Character::lore) {
            quiet.extend(lore.father.iter().cloned());
            quiet.extend(lore.mother.iter().cloned());
            quiet.extend(lore.children.iter().cloned());
        }
        // Anyone behind the same door. `HOUSEHOLD_EPSILON_M = 0.5` is below the
        // measured 1.2748 m minimum separation between any two of the 413 baked
        // doors, so this can only ever fire for a generated citizen sharing one
        // `nav::Door` under the occupancy cap — never for two of the cast.
        if let Some(door) = world.household_doors.get(who) {
            for (other, at) in world.household_doors.iter() {
                if at.distance(*door) <= HOUSEHOLD_EPSILON_M {
                    quiet.insert(other.clone());
                }
            }
        }
    }
    let craft_ear = subject
        .first()
        .and_then(|who| world.characters.get(who))
        .and_then(Character::lore)
        .and_then(|lore| lore.occupation_id.clone());
    (quiet, craft_ear)
}
```

`world.household_doors.iter()` is O(doors) per subject; a fact has 1–2 subjects and is born once, so
at 20,000 housed bodies that is ~40,000 distance tests **per mint**, not per poll. Fine, and stated
so nobody moves it into the hot path.

In `FactCatalog::seed` (M1), after the subject ids resolve, add the two fields to the `Fact` literal
from `let (quiet_among, craft_ear) = crate::knowledge::frozen_ears(world, &subject);` — M1's `Fact`
has neither field (M1 step 6's scope note), so this is an addition, not a replacement.

```rust
//! The transport half: the ward's air, the rolls that move a word into a mouth,
//! and the instrument that measures how fast (`../../../../features/knowledge_and_rumor/02_rumor_pollen.md`).

use std::{
    collections::{BTreeMap, BTreeSet},
    hash::{DefaultHasher, Hash, Hasher},
    sync::OnceLock,
};

use crate::{
    areas::AreaMap,
    character::Character,
    clock::WorldClock,
    ids::{ActorId, AreaKey, FactId, FactKey},
    knowledge::{
        AIR_PER_WARD_MAX, GARBLE_AREA_RADIUS_M, HEAT_GONE_BELOW, HOP_LOSS, PLAYER_CURIOSITY,
        STIRS_PER_GAME_HOUR, VOLUNTEER_HEAT, WARD_CELL_M, Fact, FactView, Held, Telling, Topic,
        heat_pct,
    },
    lore::PlanningWard,
    math::Vec3,
    world::World,
};

/// The walkable city's own world box, read off `assets/world/navigation.json`'s
/// `grid` (`x0 -365.0`, `z0 -480.5`, `cell_m 0.25`, `w 2907`, `h 3313`) →
/// 726.75 × 828.25 m. **Not the door hull** (x ∈ [-323.6, 329.1],
/// z ∈ [-457.1, 324.9]): a grid sized to the doors leaves real city paving
/// unanswered, and a person standing on it would carry and deposit nothing.
const CITY_MIN_X: f64 = -365.0;
const CITY_MIN_Z: f64 = -480.5;
const CITY_SPAN_X: f64 = 726.75;
const CITY_SPAN_Z: f64 = 828.25;

/// A cell whose centre and corners disagree. 8 wards means ordinals 0..=7, so
/// this can never collide with one.
const AMBIGUOUS: u8 = 0xFF;

/// Position → standing ward, exactly, in one array index most of the time.
///
/// An **accelerator, not an approximation**. A cell holds a ward ordinal only
/// where its centre *and all four corners* agree under the exact nearest-mark
/// search, and [`AMBIGUOUS`] otherwise; an ambiguous cell — and any point
/// outside the city box — falls through to that same exact search. So the
/// grid's answer is identical to the exact search *everywhere*, provably,
/// because the fallback **is** the exact search. That is what lets `crowd.rs`
/// be routed through it without moving anybody's house (D23).
#[derive(Debug, Clone, PartialEq)]
pub struct WardGrid {
    cells: Vec<u8>,
    cols: usize,
    rows: usize,
    /// The 321 ward-labelled baked doors, in `homes::ward_marks()`'s sorted
    /// order. Held here so the exact search never re-parses `homes.json`.
    marks: Vec<([f64; 2], PlanningWard)>,
}

/// The grid, baked once per process: 91 × 104 = 9,464 bytes plus the marks.
pub fn ward_grid() -> &'static WardGrid {
    static GRID: OnceLock<WardGrid> = OnceLock::new();
    GRID.get_or_init(WardGrid::bake)
}

impl WardGrid {
    pub fn bake() -> Self {
        let marks = crate::crowd::ward_map();
        let cols = (CITY_SPAN_X / WARD_CELL_M).ceil() as usize;
        let rows = (CITY_SPAN_Z / WARD_CELL_M).ceil() as usize;
        let mut cells = vec![AMBIGUOUS; cols * rows];
        if marks.is_empty() {
            return Self { cells, cols, rows, marks };
        }
        for cx in 0..cols {
            for cz in 0..rows {
                let x = CITY_MIN_X + cx as f64 * WARD_CELL_M;
                let z = CITY_MIN_Z + cz as f64 * WARD_CELL_M;
                let centre = exact_over(&marks, x + WARD_CELL_M / 2.0, z + WARD_CELL_M / 2.0);
                let agreed = [(0.0, 0.0), (WARD_CELL_M, 0.0), (0.0, WARD_CELL_M),
                              (WARD_CELL_M, WARD_CELL_M)]
                    .iter()
                    .all(|(dx, dz)| exact_over(&marks, x + dx, z + dz) == centre);
                if agreed && let Some(ward) = centre {
                    cells[cz * cols + cx] = ordinal(ward);
                }
            }
        }
        Self { cells, cols, rows, marks }
    }

    /// **The** definition of where a person is standing.
    pub fn at(&self, point: Vec3) -> Option<PlanningWard> {
        let cx = ((point.x - CITY_MIN_X) / WARD_CELL_M).floor();
        let cz = ((point.z - CITY_MIN_Z) / WARD_CELL_M).floor();
        if cx >= 0.0 && cz >= 0.0 {
            let (cx, cz) = (cx as usize, cz as usize);
            if cx < self.cols && cz < self.rows {
                let cell = self.cells[cz * self.cols + cx];
                if cell != AMBIGUOUS {
                    return Some(PlanningWard::ALL[usize::from(cell)]);
                }
            }
        }
        self.exact_at(point)
    }

    /// The exact nearest-mark search — `crowd::nearest_ward`'s body, moved here
    /// so there is one copy of it. The tie-break is strictly `<`, so the answer
    /// depends on `homes::ward_marks()`' sort order and that sort is therefore
    /// load-bearing (`homes.rs:105`).
    pub fn exact_at(&self, point: Vec3) -> Option<PlanningWard> {
        exact_over(&self.marks, point.x, point.z)
    }

    pub fn is_empty(&self) -> bool { self.marks.is_empty() }
    pub fn cells(&self) -> usize { self.cells.len() }
    pub fn ambiguous_cells(&self) -> usize {
        self.cells.iter().filter(|cell| **cell == AMBIGUOUS).count()
    }
}

fn exact_over(marks: &[([f64; 2], PlanningWard)], x: f64, z: f64) -> Option<PlanningWard> {
    let mut best: Option<(f64, PlanningWard)> = None;
    for ([mx, mz], ward) in marks {
        let (dx, dz) = (mx - x, mz - z);
        let distance = dx * dx + dz * dz;
        if best.is_none_or(|(closest, _)| distance < closest) {
            best = Some((distance, *ward));
        }
    }
    best.map(|(_, ward)| ward)
}

/// Declaration order, which is `PlanningWard::ALL`'s order (`lore.rs:66`).
fn ordinal(ward: PlanningWard) -> u8 {
    PlanningWard::ALL.iter().position(|other| *other == ward).expect("ALL is exhaustive") as u8
}
```

```rust
impl AreaAdjacency {
    /// Five neighbours at most, inside [`GARBLE_AREA_RADIUS_M`], measured from
    /// the arithmetic mean of the area's boxes' centres. Six are asked for and
    /// the area itself dropped, because `nearest_areas` returns the query's own
    /// area at distance 0.

        for (index, area) in map.areas.iter().enumerate() {

            let mut near: Vec<AreaKey> = map
                .nearest_areas(centre, GARBLE_AREA_RADIUS_M, 6)
                .into_iter()
                .filter(|found| found.area.id != area.id)

                .collect();
            near.truncate(5);
            debug_assert!(u16::try_from(index).is_ok());
            neighbours.push(near);
        }
        Self { neighbours }
    }

        self.neighbours.get(usize::from(area.0)).map_or(&[], Vec::as_slice)
    }
}

}
```

```rust
/// `may_carry` first (D51): the subject's chance is 0.0 whatever the table says,
/// under every measurement lever. The player is rolled at [`PLAYER_CURIOSITY`]

    if !crate::knowledge::may_carry(fact, listener) {
        return 0.0;
    }
    let curiosity = if world.player_id() == Some(listener) {
        PLAYER_CURIOSITY
    } else {
        crate::attention::curiosity_of(world, listener)
    };

}

/// The one roll, and it is a hash of stable inputs and never a fresh draw —
/// `notices::carries`' idiom (`notices.rs:419`) with the two guards
/// `custody::struggle_roll` has and `carries` lacks (`custody.rs:1031`).
///
/// Hashed over `stir` and **not** over the clock: that is what stops a 1-in-20
/// chance becoming a certainty within a few seconds of polling.
///
/// Not `world::hash01` (`world.rs:1313`), which takes only `(salt, id, epoch)`:
/// the **`fact.sequence` term is load-bearing**, not for the probability — it
/// contributes nothing to it — but to decorrelate the rolls, so the same mouths
/// do not pick up everything in the air at once (`02_rumor_pollen.md`,
/// "Salience", on what the `fact` term is for).
pub fn picks_up(fact: &Fact, actor: &ActorId, stir: u32, chance: f64) -> bool {
    if chance.is_nan() || chance <= 0.0 {
        return false;
    }
    if chance >= 1.0 {
        return true;
    }
    let mut hasher = DefaultHasher::new();
    "pollen_pickup".hash(&mut hasher);
    fact.sequence.hash(&mut hasher);
    actor.hash(&mut hasher);
    stir.hash(&mut hasher);
    ((hasher.finish() >> 11) as f64 / (1u64 << 53) as f64) < chance
}
```

### 12. `knowledge/salience.rs` — `salience()`, the whole of it

```rust
/// `base(topic) × affinity(listener) × household damping` — the pure table
/// product and nothing else. **Never decays** — heat answers *is this current*,
/// this answers *is this worth repeating at all* — and what falls out of
/// multiplying them is that a cold scandal out-travels a fresh squabble.
///
/// Not `0.0` for the subject: that is a rule, `knowledge::may_carry` (D51), and
/// it sits outside this product so `SalienceTable::flat()` is exactly a
/// multiplication by one. The player has no lore and is **not** the no-trade
/// quarter: their affinity is 1.0 on every topic, so `PLAYER_CURIOSITY` is the
/// whole of their roll (D26).
///
/// Sanitised with `f64::max`/`min` chains and never `clamp`, so a NaN in the
/// table sanitises to the floor instead of propagating through every roll in the
/// city (`marks.rs:752`'s reason, in this crate's own idiom).
pub fn salience(world: &World, fact: &Fact, listener: &ActorId) -> f64 {
    let table = &*world.salience;
    let mut value = table.base(fact.topic);

    let occupation = world
        .characters
        .get(listener)
        .and_then(Character::lore)
        .and_then(|lore| lore.occupation_id.as_deref());
    let is_player = world.player_id() == Some(listener);

    value *= match (fact.topic, occupation) {
        _ if is_player => 1.0,
        // The subject's own trade hears a spoiled batch; nobody else does.
        // `craft_ear` is `None` when the subject has no trade, and then every
        // *traded* listener is the other trade.
        (Topic::Craft, Some(trade)) if fact.craft_ear.as_deref() == Some(trade) => {
            table.craft_own()
        }
        (Topic::Craft, Some(_)) => table.craft_other(),
        // The no-trade quarter hears everything — on every topic, `Craft`
        // included (`02_rumor_pollen.md`: "any topic | no occupation at all |
        // 1.4"): they have no round, they loiter where they were stood, and
        // this row is where "the beggars know everything before anyone" lives.
        (_, None) => table.no_trade(),
        (topic, Some(trade)) => {
            let (ears, multiplier) = table.ear_of(topic);
            if ears.iter().any(|ear| ear == trade) { multiplier } else { 1.0 }
        }
    };

    // The last people to hear a scandal are the ones who live with it.
    if fact.quiet_among.contains(listener) {
        value *= table.household();
    }
    value.max(0.0).min(4.0)
}
```

ear multiplier, because the match arms are exclusive — which is what keeps the worst per-roll chance

```rust
        return;

    // exceeds 15%).
    let mut deposits: Vec<(FactKey, u8, f32)> = Vec::new();
    for (key, held) in crate::knowledge::holdings_of(world, actor) {
        let Some(fact) = world.knowledge.fact(key) else { continue };
        // `may_carry` and `!fact.decays` are inside `volunteers` (M1, D51/D52).
        if !crate::knowledge::volunteers(world, fact, actor, &held, Some(game_days)) {
            continue;
        }
        deposits.push((key, held.hops, held.heat(Some(game_days))));
    }

    // ── phase 1b: one roll per fact in this ward's air
    let mut tellings: Vec<(FactKey, Telling)> = Vec::new();
    for (key, drift) in world.knowledge.ward_air(ward) {
        let Some(fact) = world.knowledge.fact(key) else { continue };
        // The subject never rolls (D51) — stated here as well as inside
        // `pickup_chance`, so a reader does not have to know it returns 0.0.
        if !crate::knowledge::may_carry(fact, actor) {
            continue;
        }
        let hops = drift.hops.saturating_add(1);
        if crate::knowledge::holds_key(world, actor, key).is_some_and(|have| have.hops <= hops) {
            continue;
        }
        // Rolled against the air's own heat; the hop loss is charged **once**,
        // at the hop, so it lands in `Telling::heat` and not in the chance
        // (`02_numbers.md` §2). Getting these two the wrong way round makes
        // every warm life and every closed form wrong.
        let chance = pickup_chance(world, fact, actor, drift.heat);
        if !picks_up(fact, actor, drift.stir, chance) {
            continue;
        }
        tellings.push((key, Telling {
            hops,
            from: drift.via.clone(),
            heat: drift.heat * HOP_LOSS,
            // M3 replaces this with `garble::view_for(world, fact, actor, hops)`.
            view: FactView::default(),
        }));
    }

    // ── phase 2: the writes
    for (key, hops, heat) in deposits {
        world.knowledge.deposit(ward, key, hops, heat, actor, game_days);
    }
    for (key, telling) in tellings {
        crate::knowledge::learn(world, actor, key, telling, Some(game_days));
    }
}

/// The stir grid's ordinal at an instant — the same half-game-hour grid the
/// sweep bumps `stir` on (D28), so a row created now starts on the current
/// coin and not on coin 0. M3's `hop_on_stage` rolls on it too.
pub(crate) fn stage_stir(game_days: f64) -> u32 {
    (game_days.max(0.0) * 24.0 * STIRS_PER_GAME_HOUR) as u32
}

/// Cool the air, bump `stir`, evict what has gone, and hold each ward to
/// `AIR_PER_WARD_MAX`.
///
/// Self-gating on the stir grid via `Knowledge::take_stir_beat`, which is
/// `marks::sweep`'s shape including its keep-the-clock-current early-out.
/// Two-phase: `Drift` is not `Copy`.
///
/// Returns whether anything moved — for tests and for `--trace-pollen` only.
/// **It must never reach `touch_public_state`**: facts are prompt state, like
/// notices (`actions.rs:3199` says so out loud), and the snapshot's 160 KiB
/// bound has 26,815 bytes left.
pub fn sweep(world: &mut World, game_days: f64) -> bool {
    if !world.knowledge_enabled {
        return false;
    }
    let previous = world.knowledge.last_sweep_game_days();
    if !world.knowledge.take_stir_beat(game_days) {
        return false;
    }
    // On the very first beat nothing has aged: the air was just deposited.
    let elapsed_hours = if previous.is_finite() {
        ((game_days - previous) * 24.0).max(0.0)
    } else {
        0.0
    };
    let factor = lambda().powf(elapsed_hours) as f32;

    let mut changed = false;
    let mut gone: Vec<(PlanningWard, FactKey)> = Vec::new();
    {
        // One `Arc::make_mut` for the whole sweep, not one per row: a per-row
        // `get_mut` would deep-copy the map on every call while any staged
        // `World` clone is alive (`inventory.rs:1298`).
        let air = world.knowledge.air_mut();
        for (key, drift) in air.iter_mut() {
            let before = heat_pct(drift.heat);
            drift.heat *= factor;
            drift.stir = drift.stir.wrapping_add(1);
            if heat_pct(drift.heat) != before {
                changed = true;
            }
            // `HEAT_GONE_BELOW` is **0.01 and never 0.0**: exponential decay
            // underflows to exactly zero and `0.0 < 0.0` is false forever, which
            // makes the row immortal and holds a cap slot nothing can reclaim —
            // the bug `MarkCatalog::from_json` refuses in as many words
            // (`marks.rs:505`).
            if drift.heat < HEAT_GONE_BELOW {
                gone.push(*key);
            }
        }
        for key in &gone {
            air.remove(key);
        }
    }
    changed |= !gone.is_empty();
    // The per-ward cap, coldest out: `heat_pct` ascending, then most hops, then
    // highest `FactKey` — the same total order the holdings cap evicts by, so
    // there is one eviction rule in the feature and it is reproducible.
    changed |= world.knowledge.trim_air_to_cap();
    changed
}
```

`air_mut` (one `Arc::make_mut`, returning `&mut BTreeMap<(PlanningWard, FactKey), Drift>`) and
`trim_air_to_cap` are `pub(crate)` on `Knowledge`.

> **A clarification on D27, not a conflict.** D27 says "every air sweep is two-phase", because
> `Drift::via` is `Option<ActorId>` and `Drift` is therefore not `Copy`. The reason is exact and the
> conclusion is one case too wide: a pass is two-phase when its *decision* needs `&World` while its
> *write* needs `&mut World`, which is `poll_person`'s pickup arm (step 13) and will be M3's
> `hop_on_stage`. Cooling needs nothing from the world but a factor, so it holds one `&mut` to the map
> and iterates `values_mut()` — which is both simpler and one `Arc::make_mut` instead of N. What must
> not be done, in either shape, is clone the air map per pass: that is a 20 Hz allocation.

`heat_pct` before **any** change test is not optional (D30): one cooling step multiplies heat by
~0.9439, twelve orders of magnitude above epsilon, so a raw-`f32` `!=` is true on every sweep and
would churn the diagnostic chain forever.

```rust
/// Deposit, pick up and cool for the whole cast, each on their own game-time
/// deadline.
///
/// A separate pass rather than a rung of `run_ladder` (`round.rs:7400`) because
/// `run_ladder` `continue`s past whole cohorts *before* its throttle — anyone
/// with a live food errand, anyone the escort abandon touched — and after it,
/// every non-`Idle`/non-`Travelling` well phase and every held conversation.
/// That is the market crowd, the well queue and the people standing talking: the
/// most gossip-shaped places in the city, all invisible from there.
///
/// It lives in `round.rs` because `Round::people` is private (`round.rs:1134`)
/// and `Round`'s only `pub(crate)` method is `abandon_bodily_errands`; a pass in
/// `knowledge::` would need three new accessors on it. Shaped like `decay_needs`
/// (`round.rs:6259`), the proven whole-cast borrow, and it calls out to
/// `knowledge::pollen` for everything but the iteration — exactly as `round.rs`
/// already reaches `crate::marks::binding_mark_about`.
fn tick_pollen(round: &mut Round, world: &mut World, clock: &WorldClock, now: f64) {
    if !world.knowledge_enabled || world.knowledge.is_empty() {
        return;
    }
    let game_days = clock.game_days(now);
    // Disjoint field borrows: `people` immutably while `next_pollen` is written.
    // A `Vec` of due ids would clone an `ActorId` (a `String`) per due person
    // per tick instead.
    let Round { people, next_pollen, .. } = round;
    for (id, person) in people.iter() {
        if next_pollen.get(id).is_some_and(|due| game_days < *due) {
            continue;
        }
        if !world.is_present(id) {
            continue;
        }
        crate::knowledge::pollen::poll_person(world, id, game_days);
        next_pollen.insert(id.clone(), game_days + poll_gap_game_days(id, person.epoch));
    }
}

/// A person's own gap, `POLLEN_POLL_GAME_MINUTES` jittered ×0.5..1.5 off their
/// id — so pickups trickle across an office instead of arriving in a column,
/// the way `leg_lag_share` already staggers their legs.
///
/// The maximum, `POLLEN_POLL_MAX_GAME_MINUTES = 15`, must stay under
/// `60 / STIRS_PER_GAME_HOUR = 30` or a person could skip a whole stir window
/// and the effective roll rate would become a function of the jitter rather
/// than the constant `S = 2.0` every closed form in `02_numbers.md` uses.
/// Asserted by `the_poll_gap_cannot_skip_a_stir`.
///
/// **Game-days out, and the epoch is the round's own** — `crate::world::hash01`
/// (`world.rs:1313`) is the crate's deterministic stand-in for an RNG and
/// `decision_jitter` (`round.rs:8820`) is the shape this copies. Salting on
/// `epoch` (`Townsperson::epoch: u64`, `round.rs:940`) means a person's gap
/// re-draws when their ladder epoch does, so nobody is stuck on the same phase
/// of the stir grid for a whole run.
fn poll_gap_game_days(id: &ActorId, epoch: u64) -> f64 {
    let unit = crate::world::hash01("pollen_poll", id, epoch);
    (POLLEN_POLL_GAME_MINUTES * (0.5 + unit)) / 1440.0
}
```

`0.5 + unit ∈ [0.5, 1.5)`, so the gap is 5..15 game minutes and
`POLLEN_POLL_MAX_GAME_MINUTES = 15.0` is the real bound rather than a stated one.

Register it in `round::tick` **immediately after** `decay_needs(round, world, clock, now);`
(**`round.rs:5741`**):

```rust
    decay_needs(round, world, clock, now);
    // The ward's air (`features/knowledge_and_rumor/`). After `decay_needs`
    // because that is where the game-days anchor is already read, and before
    // anything that moves anybody: a person deposits into the ward they are
    // standing in when the tick begins.
    tick_pollen(round, world, clock, now);
```

**The composed step invariant, which is what actually protects the roll rate.** `round::tick` runs at
most once per `MOVEMENT_TICK_SECONDS` (20 Hz, `engine.rs:1306-1309`), so the worst realised gap
between one person's polls is `POLLEN_POLL_MAX_GAME_MINUTES` plus one host step in game minutes.
Under `--watch-clock` the default step is `seconds_per_day / 200` real seconds — i.e. **7.2 game
minutes at any `seconds_per_day`**, since `24 × 60 / 200 = 7.2` — and **0.4 s (0.16 game min at the
shipped clock)** once `--trace-pollen` caps it (step 19). Under the `T` key's 60× one round tick is
1.2 game minutes. So the worst cases are `15 + 7.2 = 22.2`, `15 + 0.16 = 15.16` and `15 + 1.2 = 16.2`
game minutes, all `< 30`: no stir window can be skipped anywhere. Put that arithmetic in
`poll_gap_game_days`' doc comment.

```rust
                           own: BTreeMap<ActorId, String>, subject: Vec<ActorId>,
                           seeded: BTreeSet<ActorId>, at: Vec3, garble: GarbleMask, decays: bool,
```

```rust
/// The one thing no instant can answer: how often people change wards, and how
/// often a **holder** leaves the ward a fact was minted in.
///
/// Held by whoever is sampling — the headless tracer, the cadence test — and
/// never by `World`, so a production run pays nothing for it. The per-fact half
/// runs only for the keys it was built with.
///
/// It **undercounts** when somebody crosses two wards between samples. Sample at
/// least twice a game hour: at the shipped clock a walker covers ~105 m in half
/// a game hour against ward diameters of 200–300 m, so a skipped ward is
/// unlikely rather than merely unmeasured.
pub struct CrossingTally { /* private */ }

impl CrossingTally {
    pub fn new(watching: &[FactKey]) -> Self;
    pub fn sample(&mut self, world: &World, game_days: f64);
    /// Every accumulator at this instant. Taken **per sample** and stored beside
    /// that sample's census, so a test reads the office boundary and the game
    /// day off one run — and takes `X` from the final snapshot only, never from
    /// an office sample (the Dayspring tide annualised is ~2× the daily figure).
    pub fn snapshot(&self) -> TallySnapshot;
}

/// `01_api.md`'s `TallySnapshot`: `game_days`, `span_game_days`,
/// `per_person_per_game_day` (the city mean: total ward-boundary changes ÷
/// bodies ÷ elapsed game days), `ward_exit_rate_per_game_day` (per ward:
/// exits ÷ person-game-hours spent in it × 24 — **the** `X_W` every expectation
/// uses, D54), `warm_mint_ward_game_hours` (per watched key: Σ over samples of
/// (holders warm on it **and standing in its mint ward**) × the sample gap in
/// game hours — the closed form's own integrand), `holder_exits` and
/// `holder_exits_same_trade` (per watched key: realised exits of the mint ward
/// by anyone holding it, warm or cold — a **diagnostic**, never asserted, D54).

impl PollenCensus {
    pub fn summary(&self) -> String;

    /// runs diff cleanly and `--pollen-flat` can be compared to

    /// `holder_exits`, `crossings_per_person_per_game_day` and the per-ward rates.
    pub fn fill(&mut self, tally: &TallySnapshot);
}

/// A census of one instant. Pure: it never writes the store.
/// `TopicRow::wards_reached` is **the number of wards whose air holds a row for
/// the fact** (`ward_air` non-empty for it) — deposits, never holders' wards
/// (D54).
```

`sample` is one pass over `world.characters`, and for each body: `world.ward_at(position)`, compare
with the stored ward, and on a change bump the city total, bump `exits[previous ward]`, and — for
each watched key whose `mint_ward` equals the ward being **left** — bump `holder_exits[key]` when

`occupation_id` equals that fact's `craft_ear`. Add `gap_game_hours` to `person_game_hours[current

ward **and** pass `knowledge::volunteers`, and add `count × gap_game_hours` to
`warm_mint_ward_game_hours[key]`. Per-fact work therefore happens only for the keys the tally was
built with.

`seed_pollen_pack`'s rule, stated once and fully determined:

1. Resolve `place` against `nav.places()` (`nav/mod.rs:365`) — the first whose `name` contains
   `place` case-insensitively, in graph order. No match: `Err` naming the query and the three nearest
   names by string distance. `at = nav.node_point(place.node)`.
2. `mouths` = every `control().is_llm()` body, ordered by distance from `at` then by id — which is
   `world.characters_within(at, 1_200.0, None)`'s own order, filtered. The radius is finite on
   purpose: `neighbours_by_distance` asserts `radius.is_finite()` (**`world.rs:485`**), and 1,200 m
   is past the 727 × 828 m city's diagonal, so it is "everybody" without being infinity.
3. `seeded` = the first **`PACK_MOUTHS = 4`** of `mouths` (`02_numbers.md` §4's K). Fewer than four:
   `Err`, with the count — a pack with two mouths measures a different city.
4. `subject` = the first body **after** those four whose `occupation_id` is `Some(trade)` with
   `trade` held by none of the four; failing that, the first with `occupation_id: None`; failing
   that, `Err`. This is what makes the `Craft` fact **off-affinity for all four seeded mouths**, and
   therefore what makes `s = 0.20 × 0.60 = 0.12` and the slow end's `0.057` reproduce.
5. For each `Topic::ALL` in order, `mint::plant_for_measurement` with
   `id = FactId::from_raw(format!("pollen.{}", topic.as_str()))`,
   `said = "{subject} — a {topic} matter at {place} {day}"` with `{topic}` substituted by the
   topic's own `as_str()` before install, `GarbleMask::default_for(topic)` (`ALL`, M1 step 6), and

`EngineConfig`: `fake_mode: true`, `nav: Some(...)`,
`clock: WorldClock::new(3600.0, Office::Dayspring, 0, night_brightness)` (`clock.rs:349` — there is

`MAX_MOVEMENT_CATCHUP_SLICES × MOVEMENT_TICK_SECONDS`, the most walking one poll realises): 9,000
polls per game day, seconds of CPU at `--extra-ambient 0`.

One shared helper drives a run:

```rust
/// Which measurement levers a run is under. Two `bool`s rather than two
/// functions, so the identity test runs the *same* code path twice.
#[derive(Clone, Copy, Default)]
struct Levers { flat: bool, no_salience: bool }

/// Plant the pack at a place at `now = 0` (the Dayspring bell), then step the
/// engine at 0.4 s to `until_game_hours`, sampling census + tally snapshot
/// **twice a game hour** — one sample per stir, so a ward transit (~1.2 gh at
/// walking pace over a 200–300 m ward) cannot fall between two samples. Returns
/// every sample's census (already `fill`ed from that sample's snapshot), so a
/// test can read any bell off one run. Asserts inside that the final snapshot's
/// city-mean X ≥ 2.0, with a message naming the walking/clock coupling (D22):
/// a future clock or step change fails here, not in a band.
fn run_band(place: &str, until_game_hours: f64, levers: Levers) -> Vec<(f64, PollenCensus)>;

/// The first sample at or after `game_hours` from the mint.
fn at(samples: &[(f64, PollenCensus)], game_hours: f64) -> &PollenCensus;
```

**The bells are the city's** (`clock.rs:65-76`): a run from the Dayspring bell (7:00) meets HighWick
at +5 gh, Waning at +8 gh, Lamplight at +11 gh (nightfall), the Snuffing at +14 gh. F1 is stated at
`HighWick + 1.5 gh = 6.5 gh` and `Waning + 1.5 gh = 9.5 gh`; nothing is asserted at "a mean office".

Every realised assertion reads `wards_reached` — **air rows**, i.e. deposits (D54) — and every
expectation reads the run's own integrand, `expected_crossings` (`warm_mint_ward_game_hours × X_W /
24` with `X_W` the mint ward's exit rate from the **final** snapshot — never a hardcoded 2.368, and
never an office sample's tide). Holder exits are printed and never asserted: a single deterministic
run realises one draw, and a cold holder walking home is not "the word being said" anywhere.
